MySQL

CREATE TABLE trainees ( id INT PRIMARY KEY AUTO_INCREMENT, full_name VARCHAR(120) NOT NULL, email VARCHAR(190) NOT NULL UNIQUE, phone VARCHAR(20), status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE' );

DELETE sets status = 'INACTIVE'. Do not remove the row.
